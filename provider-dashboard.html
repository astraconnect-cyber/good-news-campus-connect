🛠️ ${booking.service || "Service Request"}

Customer: ${booking.customerName || "Not provided"}

Phone: ${booking.customerPhone || "Not provided"}

Email: ${booking.customerEmail || "Not provided"}

Location: ${booking.location || "Not provided"}

Description: ${booking.description || "No description"}

Status: ✅ Approved

`; bookingMessage.appendChild(card); }); }) .catch(function(error) { console.error("Error loading approved jobs:", error); bookingMessage.innerHTML = "

❌ Unable to load approved jobs.

"; }); }
